# Better Auth Admin Plugin & Admin Dashboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Integrate Better Auth's official `admin` plugin into the application backend and database, and build a full-stack administration experience with user search, role management (`admin` / `user`), and user banning / unbanning.

**Architecture:** Extend the Drizzle Postgres schema to include Better Auth admin fields and generate migrations. Register `admin()` in the Better Auth server config and incorporate user roles/banned checks into `fast-auth` and JWT tokens. Provide initial admin bootstrapping via `ADMIN_EMAILS` and a CLI script (`pnpm db:make-admin <email>`), and implement an admin route guard (`admin.ts` middleware), an admin API composable (`useAdmin.ts`), and a clean, responsive Admin Dashboard page at `/admin`.

**Tech Stack:** Nuxt 4, Vue 3 Composition API, Better Auth 1.7+, Drizzle ORM, PostgreSQL (Neon), Tailwind CSS, Reka UI / Lucide / Hugeicons, TypeScript.

## Global Constraints
- Target workspace: `apps/web` within the monorepo.
- Monorepo package manager: `pnpm@10.33.0`.
- Preserve existing Capacitor bearer token workflow (`set-auth-token` header) and `x-auth-jwt` fast-path optimizations.
- Zero breaking changes to existing users in PostgreSQL (all new columns must have sensible defaults or be nullable).
- Admin dashboard only accessible to authenticated users with `role === 'admin'`.

```typescript
export interface FastUser {
  id: string
  name: string
  email: string
  role: string
  banned?: boolean
  createdAt: string | Date
}

export async function signJwt(user: FastUser) {
  const header = b64url(enc.encode(JSON.stringify({ alg: 'HS256', typ: 'JWT' })))
  const now = Math.floor(Date.now() / 1000)
  const payload = b64url(enc.encode(JSON.stringify({
    sub: user.id,
    name: user.name,
    email: user.email,
    role: user.role || 'user',
    banned: user.banned ?? false,
    createdAt: new Date(user.createdAt).toISOString(),
    iat: now,
    exp: now + JWT_TTL,
  })))
  const sig = await crypto.subtle.sign('HMAC', await hmacKey(), enc.encode(`${header}.${payload}`))
  return `${header}.${payload}.${b64url(sig)}`
}

export async function verifyJwt(token: string): Promise<(FastUser & { iat?: number }) | null> {
  const [header, payload, sig] = token.split('.')
  if (!header || !payload || !sig) return null
  try {
    const ok = await crypto.subtle.verify('HMAC', await hmacKey(), fromB64url(sig), enc.encode(`${header}.${payload}`))
    if (!ok) return null
    const claims = JSON.parse(new TextDecoder().decode(fromB64url(payload)))
    if (typeof claims.exp !== 'number' || claims.exp * 1000 < Date.now()) return null
    return {
      id: claims.sub,
      name: claims.name,
      email: claims.email,
      role: claims.role || 'user',
      banned: claims.banned ?? false,
      createdAt: claims.createdAt,
      iat: claims.iat,
    }
  }
  catch {
    return null
  }
}
```

```typescript
export async function requireUser(event: H3Event): Promise<FastUser> {
  return timed(event, 'auth', async () => {
    const fast = await fastUser(event)
    if (fast) {
      if (fast.banned) {
        throw createError({ statusCode: 403, statusMessage: 'Tài khoản đã bị khóa' })
      }
      return fast
    }
    const session = await useServerAuth().api.getSession({ headers: event.headers })
    if (!session) throw createError({ statusCode: 401, statusMessage: 'Bạn cần đăng nhập' })
    const { id, name, email, createdAt } = session.user
    const rawUser = session.user as Record<string, any>
    let role = rawUser.role || 'user'
    const banned = Boolean(rawUser.banned)

    if (banned) {
      throw createError({ statusCode: 403, statusMessage: 'Tài khoản đã bị khóa' })
    }

    // Auto-promote if configured in ADMIN_EMAILS
    const adminEmails = (process.env.ADMIN_EMAILS || '')
      .split(',')
      .map(e => e.trim().toLowerCase())
      .filter(Boolean)
    if (role !== 'admin' && adminEmails.includes(email.toLowerCase())) {
      role = 'admin'
      const db = useDb()
      await db.update(schema.user).set({ role: 'admin' }).where(eq(schema.user.id, id))
    }

    const user: FastUser = { id, name, email, role, banned, createdAt }
    await rememberUser(event, user)
    return user
  })
}
```

```typescript
import 'dotenv/config'
import { eq } from 'drizzle-orm'
import { schema, useDb } from '../server/db'

async function run() {
  const email = process.argv[2]?.trim().toLowerCase()
  if (!email) {
    console.error('Usage: pnpm db:make-admin <email>')
    process.exit(1)
  }

  const db = useDb()
  const [existing] = await db
    .select({ id: schema.user.id, name: schema.user.name, email: schema.user.email, role: schema.user.role })
    .from(schema.user)
    .where(eq(schema.user.email, email))
    .limit(1)

  if (!existing) {
    console.error(`User with email "${email}" not found in database.`)
    process.exit(1)
  }

  if (existing.role === 'admin') {
    console.log(`User "${existing.name}" (${existing.email}) is already an admin.`)
    process.exit(0)
  }

  await db.update(schema.user).set({ role: 'admin' }).where(eq(schema.user.id, existing.id))
  console.log(`Successfully promoted "${existing.name}" (${existing.email}) to role "admin".`)
  process.exit(0)
}

run().catch((err) => {
  console.error('Failed to make admin:', err)
  process.exit(1)
})
```

```typescript
export interface AdminUserItem {
  id: string
  name: string
  email: string
  role: string
  banned: boolean
  banReason?: string | null
  banExpires?: string | null
  createdAt: string
}

export interface ListUsersResponse {
  users: AdminUserItem[]
  total: number
}

export function useAdmin() {
  const api = useApi()

  async function listUsers(params: {
    limit?: number
    offset?: number
    searchValue?: string
  } = {}): Promise<ListUsersResponse> {
    const query = new URLSearchParams()
    if (params.limit) query.set('limit', String(params.limit))
    if (params.offset) query.set('offset', String(params.offset))
    if (params.searchValue) {
      query.set('searchField', 'email')
      query.set('searchOperator', 'contains')
      query.set('searchValue', params.searchValue)
    }

    const res = await api<any>(`/api/auth/admin/list-users?${query.toString()}`, {
      method: 'GET',
    })

    const users = (res.users || res || []).map((u: any) => ({
      id: u.id,
      name: u.name,
      email: u.email,
      role: u.role || 'user',
      banned: Boolean(u.banned),
      banReason: u.banReason ?? null,
      banExpires: u.banExpires ? new Date(u.banExpires).toISOString() : null,
      createdAt: u.createdAt,
    }))

    return {
      users,
      total: res.total ?? users.length,
    }
  }

  async function setRole(userId: string, role: string) {
    return api('/api/auth/admin/set-role', {
      method: 'POST',
      body: { userId, role },
    })
  }

  async function banUser(userId: string, banReason?: string, banExpiresIn?: number) {
    return api('/api/auth/admin/ban-user', {
      method: 'POST',
      body: {
        userId,
        banReason: banReason || undefined,
        banExpiresIn: banExpiresIn || undefined,
      },
    })
  }

  async function unbanUser(userId: string) {
    return api('/api/auth/admin/unban-user', {
      method: 'POST',
      body: { userId },
    })
  }

  return {
    listUsers,
    setRole,
    banUser,
    unbanUser,
  }
}
```
